function formatDate(val) { if (val === null || val === undefined || val === '') return ''; // 1. Xử lý trường hợp Excel trả về MÃ SỐ (Ví dụ: 26001 -> 09/03/1971) if (typeof val === 'number' || (!isNaN(val) && !String(val).includes('/') && !String(val).includes('-'))) { let num = Number(val); if (num > 1000 && num < 100000) { // Dải mã số ngày của Excel let date = new Date(Math.round((num - 25569) * 86400 * 1000)); let d = String(date.getUTCDate()).padStart(2, '0'); let m = String(date.getUTCMonth() + 1).padStart(2, '0'); let y = date.getUTCFullYear(); return `${d}/${m}/${y}`; } return String(val); } // 2. Xử lý kiểu Date của JavaScript if (val instanceof Date) { if (isNaN(val.getTime())) return ''; let d = String(val.getDate()).padStart(2, '0'); let m = String(val.getMonth() + 1).padStart(2, '0'); let y = val.getFullYear(); return `${d}/${m}/${y}`; } // 3. Xử lý dạng chuỗi let str = String(val).replace(" 00:00:00", "").trim(); if (str.includes('-')) { let parts = str.split('-'); if (parts.length === 3 && parts[0].length === 4) { return `${parts[2].padStart(2, '0')}/${parts[1].padStart(2, '0')}/${parts[0]}`; } } return str; }
